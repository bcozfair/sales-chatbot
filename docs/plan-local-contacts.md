## 1. คำถามที่ถามบ่อย: `contact_id = 0` คืออะไร

**ใช่ — มีแถว `contact_id = 0` อยู่ใน `customers_data_view` จริง 3,852 แถว แต่มันไม่ใช่ "ผู้ติดต่อ"**
วัดบนฐาน dev 2026-09-17:

| ถาม | ตอบ |
| --- | --- |
| แถว `contact_id = 0` มีกี่แถว | **3,852** (= 3,852 บริษัท · มาจาก `customers` ที่มี `contact_id = 0` จำนวนเท่ากันพอดี) |
| ในนั้นมีชื่อผู้ติดต่อกี่แถว | **0 แถว** — `contact_name IS NULL` ทั้ง 3,852 แถว |
| มี `contact_id` ติดลบไหม | **ไม่มีเลย** ทั้งใน `customers` และใน `customers_data_view` |

แถวพวกนี้เกิดจาก **Arm 1 กรอง `c.contact_id >= 0` ไม่ใช่ `> 0` โดยตั้งใจ** เพื่อให้ "บริษัทที่ยังไม่มี
ผู้ติดต่อเลยสักคน" ยังมีแถวอยู่ใน view — ถ้ากรอง `> 0` บริษัทกลุ่มนี้จะหายจากการค้นหาบริษัทไปทั้งหมด
⇒ **แถว `contact_id = 0` คือ "แถวของบริษัท" ไม่ใช่ "แถวของผู้ติดต่อ"**

แล้วทุกจุดที่ถามหา *ผู้ติดต่อ* จึงต้องกรอง `contact_id > 0` เพื่อตัดแถวบริษัททิ้ง — `CONTACT_VIEW_COLS`,
`CDV_CONTACT_SUBQ`, `getContactsByCustomerId`, `getRelatedContactsByCustomerId`,
`ODOO_EXPORT_SALES_TEAM_JOIN`, `idx_so_contact_latest` (คอมเมนต์ที่
[repositories.ts:150](../db/repositories.ts#L150) เขียนไว้เองว่า *"contact_id>0 = ตัด row บริษัทไม่มีผู้ติดต่อ
(contact_id=0 ไม่มีชื่อ → ไม่ให้โผล่เป็นผู้ติดต่อผี)"*)

**ประโยคที่ถูกต้องกว่าที่ผมเขียนไว้รอบแรก:** `contact_id > 0` ไม่ใช่เงื่อนไข "มีตัวตนจริง" แต่เป็นเงื่อนไข
**"เป็นแถวผู้ติดต่อ ไม่ใช่แถวบริษัท"** · ผลกับการเลือกเลขเหมือนเดิมทุกประการ — **0 ถูกจองความหมาย
ไปแล้ว และเลขติดลบจะถูกตัวกรองเหล่านี้ทิ้งทั้งหมด** ⇒ เลข local ต้องเป็นบวกและสูงกว่าช่วงของ Odoo

```sql
-- contact_id ของ Odoo วิ่งอยู่แถว 1.03 ล้าน (วัด 2026-09-17: max = 1,032,141)
-- ตั้งต้นที่ 900 ล้านเพื่อกันชนโดยไม่ต้องประสานกับใคร และยังห่างเพดาน int4 (2,147,483,647) อีกเท่าตัว
-- ⚠️ ห้ามใช้ 0 หรือเลขติดลบ — 0 ถูกจองให้ "แถวบริษัทที่ไม่มีผู้ติดต่อ" (3,852 แถว ทุกแถวไม่มีชื่อ)
--    และทุกจุดที่ถามหาผู้ติดต่อกรอง contact_id > 0 เพื่อตัดแถวเหล่านั้นทิ้ง (ดู §1)
CREATE SEQUENCE IF NOT EXISTS public.local_contact_id_seq START WITH 900000001;

CREATE TABLE IF NOT EXISTS public.local_contacts (
  contact_id    integer     PRIMARY KEY DEFAULT nextval('public.local_contact_id_seq'),
  company_id    integer     NOT NULL,   -- ต้องมีจริงใน customers_data_view (ตรวจฝั่ง server ไม่ใช่ FK)
  contact_name  text        NOT NULL,   -- trim แล้วตั้งแต่ตอนบันทึก — ดู §7.3
  job_position  text,                   -- ปลายทาง = ไฟล์ export ให้แอดมินคีย์ใน Odoo (§5.3)
  contact_phone text,
  contact_email text,
  -- ⚠️ ไม่มีคอลัมน์ทีมขายที่นี่โดยตั้งใจ — ทีมขายสืบทอดจากบริษัทตอนอ่าน ไม่เก็บซ้ำ (§3.5)
  created_by    integer,                -- admin_users.id · ไม่มี FK (ลบแอดมินแล้วหลักฐานต้องอยู่)
  created_at    timestamptz NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at    timestamptz NOT NULL DEFAULT CURRENT_TIMESTAMP,

  -- ── สามคอลัมน์นี้ "ระบบเขียน" เท่านั้น ไม่มี endpoint ให้คนกด (§6) ──
  odoo_matched_at         timestamptz,  -- เจอตัวจริงใน Odoo แล้วเมื่อไหร่
  odoo_matched_contact_id integer,      -- contact_id จริงฝั่ง Odoo ที่จับคู่ได้
  odoo_matched_by         text,         -- สัญญาณไหนเป็นคนยืนยัน

  CONSTRAINT local_contacts_name_not_blank CHECK (btrim(contact_name) <> ''),
  -- ช่วงเลขเป็นข้อตกลงที่โค้ดหลายที่พึ่งพา (isLocalContactId) จึงบังคับที่ฐาน ไม่ใช่แค่ที่ sequence
  CONSTRAINT local_contacts_id_range CHECK (contact_id >= 900000000),
  CONSTRAINT local_contacts_matched_by_check CHECK (
    odoo_matched_by IS NULL OR odoo_matched_by IN ('contact_sync', 'imported_order'))
);

CREATE INDEX IF NOT EXISTS idx_local_contacts_company ON public.local_contacts (company_id);
CREATE INDEX IF NOT EXISTS idx_local_contacts_pending
  ON public.local_contacts (created_at) WHERE odoo_matched_at IS NULL;

-- ชื่อซ้ำในบริษัทเดียวกันต้องกันที่ฐาน ไม่ใช่แค่ที่ endpoint — ไม่งั้นกดปุ่มรัว ๆ สองครั้งได้สองแถว
-- (เทียบแบบ btrim ให้ตรงกับเกณฑ์ dedupe ของ Arm 3 เป๊ะ)
CREATE UNIQUE INDEX IF NOT EXISTS idx_local_contacts_company_name
  ON public.local_contacts (company_id, btrim(contact_name));
```

### 3.2 Arm 3 — ⚠️ SQL ใน §5.2 **ของแผนใหญ่** รันไม่ผ่าน

**แผนใหญ่ §5.2 สั่งให้ "ต่อท้าย CTE `base` ด้วย `UNION ALL`" แต่ท่อนที่จะต่อนั้นอ่าน `base` เอง** — CTE ธรรมดา
อ้างถึงตัวเองไม่ได้ **พิสูจน์บนฐาน dev 2026-09-17:**

```
WITH base AS (SELECT 1 AS x UNION ALL SELECT b2.x FROM base b2) SELECT * FROM base
→ ERROR: relation "base" does not exist
```

รูปที่ถูกคือแยกเป็น CTE ของตัวเองแล้วรวมที่ `SELECT` ท้าย — และได้ของแถมที่สำคัญกว่าความถูกต้อง
ทางไวยากรณ์: `comp` / `own_last` / `own_credit` / `ent_keys` **ยังอ่าน `base` เหมือนเดิม** ⇒ แถว local
**ไม่มีทางขยับค่าระดับบริษัทหรือคำตอบของด่านเครดิต**ได้เลยแม้แต่บิตเดียว

```sql
CREATE OR REPLACE VIEW public.customers_data_build AS
WITH latest_so AS ( ... คงเดิมทั้งดุ้น ... ),
base AS ( ... Arm 1 UNION ALL Arm 2 — คงเดิมทั้งดุ้น ... ),

-- ── แถวตัวแทนของบริษัทที่มีผู้ติดต่อ local (ค่าระดับบริษัทยืมจากแถวนี้) ──
-- เขียนเป็น DISTINCT ON ครั้งเดียว ไม่ใช่ LATERAL ... LIMIT 1 ต่อแถวอย่างที่แผนใหญ่ร่างไว้:
-- base ถูกอ้าง 6 ที่ ⇒ Postgres materialize มัน ⇒ LATERAL จะกลายเป็น seq scan 82,721 แถว
-- **ต่อผู้ติดต่อ local หนึ่งคน** ส่วนรูปนี้สแกนรอบเดียวแล้ว hash join
local_anchor AS (
  SELECT DISTINCT ON (b.company_id) b.*
    FROM base b
   WHERE b.company_id IN (SELECT company_id FROM public.local_contacts)
   ORDER BY b.company_id, b.contact_id
),
-- ── ทีมขาย: สืบทอดจาก "บริษัท" ไม่ใช่จากแถวตัวแทนดื้อ ๆ (เจ้าของเคาะ 2026-09-17) ──
-- หยิบค่าแรกที่ไม่ว่างเรียงตาม contact_id = กติกาเดียวกับ CTE `comp` ของคอลัมน์อื่น
-- วัด 2026-09-17: 663 จาก 53,490 บริษัท (1.2%) แถวตัวแทนมีทีมขายว่างทั้งที่พี่น้องมีค่า
--   ⇒ ถ้าใช้ค่าของแถวตัวแทนตรง ๆ ผู้ติดต่อใหม่ของ 663 บริษัทนี้จะได้ทีมขายว่างโดยไม่จำเป็น
--   · 320 บริษัท (0.6%) มีมากกว่าหนึ่งทีม ⇒ หยิบของ contact_id น้อยสุดที่ไม่ว่าง ให้ผลคงที่
-- ⚠️ บริษัทที่ไม่มีทีมขายเลย (59.1%) จะได้ NULL — **ปล่อย NULL ไว้อย่างนั้น ไม่ต้องหาค่ามาเติม** (§3.5)
-- ⚠️ ห้ามย้าย sales_team เข้า CTE `comp` เพื่อ "ให้เหมือนกัน" — `comp` มีคอมเมนต์กำกับว่าทีมขาย
--    ไม่ใช่คุณสมบัติของบริษัท (ผู้ติดต่อคนละคนอาจคนละทีม) การสืบทอดนี้ใช้ได้เฉพาะกับ **คนใหม่
--    ที่ไม่มีประวัติ** เท่านั้น ส่วนคนเดิมที่ Odoo ระบุทีมมาแล้ว ห้ามไปทับ
local_team AS (
  SELECT b.company_id,
         (array_remove(array_agg(b.sales_team ORDER BY b.contact_id), NULL))[1] AS sales_team
    FROM base b
   WHERE b.company_id IN (SELECT company_id FROM public.local_contacts)
   GROUP BY b.company_id
),
-- ── ผู้ติดต่อที่ Odoo สร้างให้แล้ว (ชื่อตรงกัน) — แถว local ต้องหลบให้ ──
-- anti-join แทน NOT EXISTS ต่อแถว ด้วยเหตุผลเดียวกับ local_anchor
local_taken AS (
  SELECT DISTINCT l.contact_id
    FROM public.local_contacts l
    JOIN base b3 ON b3.company_id = l.company_id
                AND btrim(b3.contact_name) = btrim(l.contact_name)
),
-- ── Arm 3: ผู้ติดต่อที่แอดมินเพิ่มเองจากหน้าเว็บ (ยังไม่มีใน Odoo) ──
local_rows AS (
  SELECT
    l.company_id, l.contact_id, 'local'::text AS source,
    a.customer_name, a.customer_reference, a.customer_tax_id, a.customer_payment_terms,
    a.customer_sale_area, a.salesperson,
    t.sales_team,                       -- ← ค่าระดับบริษัท ไม่ใช่ a.sales_team ของแถวตัวแทน
    a.customer_type, a.phone, a.mobile, a.email,
    public.clean_text(l.contact_name)  AS contact_name,
    public.clean_text(l.contact_phone) AS contact_mobile,
    public.clean_text(l.contact_phone) AS contact_phone,
    public.clean_text(l.contact_email) AS contact_email,
    a.invoice_street, a.invoice_district, a.invoice_sub_district, a.invoice_state, a.invoice_zip
  FROM public.local_contacts l
  JOIN local_anchor a ON a.company_id = l.company_id
  JOIN local_team   t ON t.company_id = l.company_id
  WHERE l.contact_id NOT IN (SELECT contact_id FROM local_taken)
),
all_rows AS (SELECT * FROM base UNION ALL SELECT * FROM local_rows),

comp AS ( ... FROM base ... ),   -- ⚠️ ห้าจุดข้างล่างคง `FROM base` ไว้ ห้ามเปลี่ยนเป็น all_rows
so_last AS ( ... ), own_last AS ( ... FROM base ... ),
own_credit AS ( ... FROM base ... ), ent_keys AS ( ... FROM base ... ),
key_last AS ( ... ), ent_last AS ( ... )

SELECT ...                        -- ⚠️ จุดเดียวที่เปลี่ยนใน SELECT ท้าย:
FROM all_rows b                   --    เดิม `FROM base b` → `FROM all_rows b`
LEFT JOIN comp ON ... LEFT JOIN own_last ON ...
LEFT JOIN own_credit ON ... LEFT JOIN ent_last ON ...;
```

**ทำไมแถว local ต้องผ่าน `SELECT` ท้ายตัวเดียวกัน ไม่ใช่ `UNION ALL` ต่อท้ายทั้ง view:** เพราะ
`SELECT` ท้ายเป็นที่ที่ `COALESCE(b.customer_sale_area, comp.customer_sale_area)` และคู่แฝดของมัน
(`invoice_district` / `invoice_sub_district`) ทำงาน ถ้าไม่ผ่านตรงนี้ ผู้ติดต่อใหม่จะมีที่อยู่ไม่ครบทั้งที่
คนอื่นในบริษัทเดียวกันครบ · และ `last_order_at` ก็มาจาก join ชุดนี้ ⇒ ผู้ติดต่อใหม่ของลูกค้าที่ติด
เครดิตค้างจะ **หลุดด่าน** ถ้าไม่ผ่าน

> **ข้อพิสูจน์ว่าไม่กระทบของเดิม:** `local_contacts` ว่าง ⇒ `local_rows` ว่าง ⇒ `all_rows` = `base`
> ทุกแถวทุกคอลัมน์ · เป็นด่านข้อแรกที่ต้องรัน (§8)

### 3.3 watermark — หนึ่งบรรทัดใน `refreshCustomerDirectory.ts`

```sql
SELECT GREATEST( (SELECT max(sync_updated_at) FROM public.customers),
                 (SELECT max(updated_at)      FROM public.sale_orders),
                 (SELECT max(updated_at)      FROM public.local_contacts) )   -- ← เพิ่ม
```
ไม่งั้นมีคนแก้ `local_contacts` ตรง ๆ แล้ว rebuild รอบถัดไป "ข้ามเพราะไม่มีอะไรเปลี่ยน"
⚠️ **ตัว reconcile (§6.1) ห้ามแตะ `updated_at`** — มันไม่ได้แก้ข้อมูลธุรกิจ ถ้าไปบวก watermark ทุกรอบ
จะสั่ง rebuild เพิ่มฟรี ๆ รอบละครั้ง
📌 migration ที่แก้แค่นิยาม view ต้องรัน refresh ด้วย `{ force: true }` ไม่งั้นดูเหมือนรันแล้วแต่ไม่มีอะไรเปลี่ยน

### 3.4 dual-write — ให้ใช้ได้ทันทีโดยไม่ rebuild ทั้งก้อน

`customers_data_view` เป็นตารางจริง ⇒ `INSERT` ตรงได้ · เขียน 2 ที่ใน transaction เดียว
**เขียนแถวใหม่โดย "ลอกจากแถวตัวแทน" ไม่ใช่ประกอบค่าเอง** — 24 คอลัมน์ที่ก๊อปมาไว้ในโค้ด TS คือ
24 ช่องที่รอวันไม่ตรงกับนิยาม view:

```sql
INSERT INTO public.customers_data_view                      -- คอลัมน์ครบทั้ง 24 ตามลำดับของ view
SELECT v.company_id, $2::int, 'local',
       v.customer_name, v.customer_reference, v.customer_tax_id, v.customer_payment_terms,
       v.customer_sale_area, v.salesperson,
       -- ทีมขาย: ค่าแรกที่ไม่ว่างของทั้งบริษัท — ต้องตรงกับ local_team ของ Arm 3 เป๊ะ
       -- (ด่าน §8 ข้อ 3 ตรวจ) · ไม่มีอะไรมาก่อนมันและไม่มีอะไรมาทีหลัง ว่างก็คือ NULL
       (SELECT (array_remove(array_agg(w.sales_team ORDER BY w.contact_id), NULL))[1]
          FROM public.customers_data_view w
         WHERE w.company_id = $1::int AND w.source <> 'local'),
       v.customer_type, v.phone, v.mobile, v.email,
       public.clean_text($3), public.clean_text($4), public.clean_text($4), public.clean_text($5),
       v.invoice_street, v.invoice_district, v.invoice_sub_district, v.invoice_state, v.invoice_zip,
       v.last_order_at
  FROM public.customers_data_view v
 WHERE v.company_id = $1::int AND v.source <> 'local'
 ORDER BY v.contact_id                                       -- เกณฑ์เดียวกับ local_anchor เป๊ะ
 LIMIT 1
ON CONFLICT DO NOTHING;
```

`source <> 'local'` ทั้งสองที่ = ลอกจากแถวที่ Odoo เป็นเจ้าของเท่านั้น (ไม่ลอกต่อจากผู้ติดต่อ local
ที่เพิ่มไปก่อนหน้า ซึ่งจะทำให้ค่าเพี้ยนสะสม) · **ด่านข้อ 3 ใน §8 เป็นตัวพิสูจน์ว่า `INSERT` นี้กับ Arm 3
ให้แถวเดียวกันทีละคอลัมน์ทั้ง 24 ช่อง** — ถ้าวันหนึ่งมีคนแก้ฝั่งเดียว ด่านนั้นล้มทันที

**⚠️ ช่องว่างเดียวที่เหลือ: build+swap กำลังทำงานพอดี** — ถ้ารอบ rebuild เริ่ม `CREATE TABLE AS`
ไปแล้ว**ก่อน**ที่เราจะ commit แถวใหม่จะตกอยู่ในตารางเก่าที่กำลังจะถูก DROP ⇒ ผู้ติดต่อหายจนกว่าจะ
rebuild รอบหน้า (≤10 นาที) กว้างประมาณ **ความยาวของ build (~2–3 วิ)**
ปิดด้วย `ensureDirectoryRow(contactId)` (คือ `INSERT ... ON CONFLICT DO NOTHING` ข้างบน ซึ่ง
idempotent อยู่แล้ว) เรียก **2 จุด**: หลัง COMMIT ของการเพิ่ม · และต้นทาง `createDraft()` เมื่อ
`isLocalContactId(contact_id)` — จุดหลังคือจุดที่ "หาย" แล้วเจ็บจริง (snapshot ของใบจะไม่มีชื่อผู้ติดต่อ)

### 3.5 ทีมขาย — สืบทอดอย่างเดียว · ไม่โผล่บนจอ · ว่างก็ปล่อยว่าง (เจ้าของเคาะ 2026-09-17)

**ทีมขายไม่ใช่เรื่องของโมดูลนี้** — มันเป็นแค่คอลัมน์หนึ่งของ `customers_data_view` ที่แถว local
ต้องมีค่าให้ครบเหมือนแถวอื่น ไม่ใช่ข้อมูลที่แอดมินกรอกตอนเพิ่มคน · **กติกาจึงสั้น 3 ข้อ**

1. **สืบทอดจากบริษัทตอนอ่าน ไม่เก็บซ้ำในตาราง** — ไม่มีคอลัมน์ทีมขายใน `local_contacts` เลย
   (§3.1) ค่าเกิดจาก CTE `local_team` ของ Arm 3 ทุกครั้งที่ rebuild ⇒ วันที่ Odoo ให้ทีมขายกับบริษัทนั้น
   ผู้ติดต่อที่เราเพิ่มไว้ **ตามไปเองโดยไม่ต้องมีใครแก้อะไร** ถ้าเก็บสำเนาไว้ มันจะแช่แข็งอยู่อย่างนั้น
   แล้วไม่มีใครรู้ว่าทำไมไม่ขยับ
2. **ไม่มีช่องบนจอ ไม่มีปุ่มแก้ ไม่มี endpoint** — ไม่ต้องมี `listSalesTeams()` ไม่ต้องมี dropdown
   ไม่ต้องตรวจว่าค่าที่ส่งมาถูกไหม เพราะไม่มีใครส่งค่ามา
3. **ได้ `NULL` ก็ปล่อย `NULL`** — **31,605 จาก 53,490 บริษัท (59.1%) ไม่มีทีมขายเลยสักแถว**
   (วัด 2026-09-17) ⇒ ผู้ติดต่อใหม่ของบริษัทกลุ่มนี้จะได้ทีมขายว่าง **และนั่นคือค่าที่ถูกต้อง
   ไม่ใช่ข้อมูลขาด** — ผู้ติดต่อเดิมของบริษัทเดียวกันก็ว่างเหมือนกันอยู่แล้ววันนี้

> **เหตุผลของเจ้าของ (2026-09-17):** *การนำเข้าผู้ติดต่อใหม่เข้า Odoo ไม่ได้ใช้ template เดียวกับ
> ไฟล์ export ใบเสนอราคา — เป็น template แยก เอาเฉพาะข้อมูลที่จำเป็นเพราะนำเข้าแบบ manual*
> ⇒ **ทีมขายไม่ได้อยู่ในไฟล์ที่คนจะเอาไปคีย์** (§5.3) การให้แอดมินกรอกจึงเป็นการเพิ่มช่องที่ไม่มี
> ปลายทาง — ตรงข้ามกับ `job_position` ที่มีปลายทางชัดว่าคนคีย์ต้องกรอกช่องไหนใน Odoo

**แล้วคอลัมน์ I ของไฟล์ *ใบเสนอราคา* ล่ะ** — เป็นคนละเรื่องและถูกปิดไปแล้วโดยเฟส H: ใบที่ยืนยัน
ตรึงทีมขายลงใบตอนนั้น (§7.6) ค่าที่ถูกตรึงคือค่าที่สืบทอดมาตามข้อ 1 · บริษัทที่ไม่มีทีมขายจะได้
คอลัมน์ I ว่าง ซึ่ง **เป็นพฤติกรรมเดิมของใบทุกใบของลูกค้ากลุ่มนั้นอยู่แล้ว ไม่ใช่ของใหม่ที่โมดูลนี้
ทำให้แย่ลง** — จะแก้ต้องไปแก้ที่ Odoo ให้บริษัทมีทีมขาย ไม่ใช่มาเติมที่ปลายทาง

**สิทธิ์:** เจ้าของเคาะ 2026-09-17 ว่า **ฟีเจอร์นี้อยู่บนหน้าเว็บเท่านั้น ไม่เปิดให้ฝั่ง LINE** เพราะคนใน
LINE คือเซลส์ ไม่ใช่แอดมิน · ทั้ง 5 endpoint จึงอยู่ใต้ `adminAuthMiddleware` และไม่มีตัวไหนที่ LIFF
เรียกได้ (`/api/customer/:id/contacts` ที่ LIFF ใช้เป็น **อ่านอย่างเดียว** เหมือนเดิม)

> ⚠️ **"เพิ่มจาก LINE ไม่ได้" ≠ "ไม่โผล่ใน LINE"** — ผู้ติดต่อที่แอดมินเพิ่มจะอยู่ใน
> `customers_data_view` ⇒ โผล่ใน picker ของ LIFF และในแชทด้วย และใบที่เซลส์ออกให้คนนั้นจาก LINE
> **ก็ติดธง `new_contact` เหมือนกัน** ซึ่งถูกแล้ว (ผู้ติดต่อก็ยังไม่มีใน Odoo อยู่ดี ไม่ว่าใบจะออกทางไหน)

> **เซลส์จะออกใบให้ผู้ติดต่อคนนั้นจาก LINE ไม่ได้เลย** — เป็นการตัดสินใจแยก ไม่ได้รวมอยู่ในเฟสนี้

**การตรวจของ `POST`** (ผลถ้าไม่ผ่านคือ `400` ทุกข้อ ยกเว้นที่ระบุ)

### 5.5 แก้ไขผู้ติดต่อ — และป้ายที่ระบบคำนวณให้เอง

**สามสถานะ สามสิทธิ์** (ระบบตัดสินจากข้อมูล ไม่ต้องให้ใครเลือก)

| สถานะของแถว | แก้ชื่อ | แก้ตำแหน่ง/โทร/อีเมล | ลบ |
| --- | --- | --- | --- |
| ยังไม่เข้า Odoo · **ยังไม่มีใบอ้างถึง** | ✅ | ✅ | ✅ |
| ยังไม่เข้า Odoo · **มีใบอ้างถึงแล้ว** | ❌ | ✅ | ❌ `409` |
| **เข้า Odoo แล้ว** (`odoo_matched_at`) | ❌ | ❌ | ❌ |

**ไม่มีแถวไหนแก้ทีมขายได้ ไม่ว่าสถานะใด** — ไม่ใช่ข้อห้าม แต่เพราะไม่มีช่องให้แก้ตั้งแต่ต้น (§3.5)

**ทำไมมีใบแล้วห้ามแก้ชื่อ:** ชื่อถูกตรึงลง snapshot ของใบและคอลัมน์ C ของไฟล์ไปแล้ว การแก้ฝั่งเรา
**ไม่ทำให้ใบที่ออกไปแล้วนำเข้าได้** — เคสนั้นต้องไปแก้ที่ Odoo ให้ตรงกับใบแทน
**ทำไมเข้า Odoo แล้วห้ามแก้:** แถว local ถูกแถวจริงบังไปแล้ว แก้ไปก็ไม่มีผลกับระบบ มีแต่จะทำให้
สองฝั่งไม่ตรงกันโดยไม่มีใครรู้ — ให้ไปแก้ที่ Odoo

**ป้ายบนแถว — แทน "ติ๊กว่าไม่ตรงกับ Odoo" ที่ต้องให้คนตัดสิน ระบบหาหลักฐานมาให้แทน**

| ป้าย | เกณฑ์ | สิ่งที่แอดมินทำต่อ |
| --- | --- | --- |
| 🟢 **เข้า Odoo แล้ว** | `odoo_matched_at IS NOT NULL` | ไม่ต้องทำอะไร — ออกจากรายการค้างเอง |
| ⚪ **รอคีย์** | ค้าง < 7 วัน | คีย์ตามไฟล์ export |
| 🟡 **ค้าง N วัน** | ค้าง ≥ 7 วัน | เช็กว่าลืมหรือคีย์แล้วชื่อไม่ตรง |
| 🔴 **น่าจะคีย์แล้วแต่ชื่อไม่ตรง** | เจอชื่อใกล้เคียงในบริษัทเดียวกัน | กด **"ใช้ชื่อนี้"** (ถ้ายังไม่มีใบอ้างถึง) หรือไปแก้ชื่อใน Odoo ให้ตรงกับใบ |

**เกณฑ์ "ชื่อใกล้เคียง" สองชั้น** — `pg_trgm` ติดตั้งอยู่แล้วบนฐานนี้ (ยืนยัน 2026-09-17)

1. **ชั้นหลัก — normalized-equal:** ตัดคำนำหน้า (`คุณ` `นาย` `นาง` `นางสาว` `K.` `khun`) + ตัดช่องว่าง
   ทั้งหมด แล้วเทียบตรงตัว
2. **ชั้นรอง — `similarity() >= 0.4`** สำหรับสะกดเพี้ยน

วัดจริง 2026-09-17 — **ชั้นเดียวไม่พอทั้งคู่ นี่คือเหตุผลที่ต้องมีสองชั้น:**

| คู่ชื่อ | `similarity()` | normalized-equal |
| --- | --- | --- |
| `คุณสมชาย ใจดี` ~ `สมชาย ใจดี` | 0.563 | ✅ |
| `K.สมชาย ใจดี` ~ `สมชาย ใจดี` | 0.846 | ✅ |
| **`คุณแนน` ~ `แนน`** | **0.222** ← ต่ำกว่าเกณฑ์ | ✅ ← ชั้นที่ 1 ช่วยไว้ |
| `สมชาย ใจดี` ~ `สมชาญ ใจดี` (สะกดเพี้ยน) | 0.692 | ❌ ← ชั้นที่ 2 ช่วยไว้ |
| `สมชาย ใจดี` ~ `นภา พรหมมา` | 0.000 | ❌ |

> ⚠️ **ใช้ `[[:space:]]` ไม่ใช่ `\s` ในการตัดช่องว่าง** — วัด 2026-09-17:
> `regexp_replace('สมชาย  ใจดี', '\s+', '', 'g')` คืนค่าเดิมไม่เปลี่ยน ส่วน `[[:space:]]+` และ `' +'`
> ทำงานถูกต้อง (backslash ถูกกลืนระหว่างทางผ่านชั้น escape) — กับดักเดียวกับที่บันทึกไว้เรื่อง heredoc

"เพิ่มงาน" — มัน**กลบกับดักข้อ 7.4 ทั้งหมด** คนที่คีย์ชื่อผิดแล้วกดติ๊กว่าเสร็จ จะทำให้ระบบเชื่อว่าพร้อม
แล้วปล่อยใบเข้าไฟล์ปกติ → ไปตกที่ Odoo โดยไม่มีใครรู้ว่าเพราะอะไร
**สัญญาณอัตโนมัติไม่มีทางโกหกแบบนั้นได้ เพราะมันวัดจากของที่ Odoo ส่งกลับมาจริง**

### 6.1 `reconcileLocalContactOdooLinks()` — ท้ายทุกรอบ sync

ทำตามแบบ `reconcileQuotationOdooLinks()` ที่มีอยู่แล้วทุกกติกา: **ห้าม throw** (ถูกเรียกท้ายรอบ sync) ·
**เขียนเฉพาะแถวที่ยังว่าง** = สถานะเดินหน้าทางเดียว รันซ้ำกี่รอบก็ไม่เปลี่ยนค่าเดิม ·
**ต้องรันหลัง `refreshCustomerDataView()`**

**สัญญาณ A — `contact_sync`: Odoo มีผู้ติดต่อคนนี้แล้ว**

```sql
UPDATE public.local_contacts l
   SET odoo_matched_at = NOW(), odoo_matched_contact_id = v.contact_id,
       odoo_matched_by = 'contact_sync'
  FROM public.customers_data_view v
 WHERE l.odoo_matched_at IS NULL
   AND v.company_id = l.company_id
   AND v.source <> 'local'
   AND btrim(v.contact_name) = btrim(l.contact_name)
```
เกณฑ์นี้**เหมือนกับ `local_taken` ของ Arm 3 ทุกตัวอักษร** ⇒ ได้ค่าคงเส้นคงวาฟรี:
**`odoo_matched_at` ถูกเขียนเมื่อไหร่ = แถว local ถูกซ่อนจาก view เมื่อนั้น** (ด่าน §8 ข้อ 7 ตรวจข้อนี้)

**สัญญาณ B — `imported_order`: ใบที่อ้างผู้ติดต่อคนนี้ถูกนำเข้า Odoo สำเร็จแล้ว** (ที่เจ้าของอธิบาย)

```sql
UPDATE public.local_contacts l
   SET odoo_matched_at = NOW(), odoo_matched_contact_id = s.contact_id,
       odoo_matched_by = 'imported_order'
  FROM public.quotations q
  JOIN public.sale_orders s ON s.sale_order_id = q.odoo_so_id
 WHERE l.odoo_matched_at IS NULL
   AND q.contact_id = l.contact_id
   AND q.odoo_imported_at IS NOT NULL
   AND s.contact_id > 0
   AND btrim(s.contact_name) = btrim(l.contact_name)   -- ⚠️ กันหยิบ partner ของบริษัทมาเป็นผู้ติดต่อ
```
`odoo_imported_at` / `odoo_so_id` ถูกเขียนไว้แล้วโดย `reconcileQuotationOdooLinks()` ท้ายรอบ sync
เดียวกัน ([quotationOdooLink.ts](../services/quotationOdooLink.ts)) ⇒ **ไม่ต้องยิง Odoo เพิ่มเลยสักครั้ง**

**ทำไมต้องมีทั้งสองสัญญาณ ไม่ใช่เลือกอันเดียว**

| | A ยิงเมื่อ | B ยิงเมื่อ |
| --- | --- | --- |
| จังหวะ | **เร็วกว่า** — ทันทีที่ Odoo มีผู้ติดต่อ แม้ยังไม่ได้ออก/นำเข้าใบ | ช้ากว่า — ต้องมีใบที่นำเข้าสำเร็จแล้ว |
| ความหนักแน่น | บอกว่า "มีคนชื่อนี้อยู่" | **บอกว่า "ใบนำเข้าได้จริง"** ซึ่งเป็นเป้าหมายตัวจริง |
| จุดบอด | **gateway ตัด `active=false` ทิ้ง** (เพดาน 5,906 contact ที่บันทึกไว้ใน CLAUDE.md) ⇒ บางคนไม่มีวันถูก sync กลับมา | ผู้ติดต่อที่ยังไม่เคยมีใบจะไม่ถูกยืนยันเลย |

⇒ A ปิดจุดบอดของ B และ B ปิดจุดบอดของ A · รัน A ก่อน B ในรอบเดียวกัน

**และถ้าคีย์ชื่อไม่ตรง ทั้ง A และ B จะไม่ยิงทั้งคู่** (B ยิงไม่ได้เพราะใบจะนำเข้าไม่ผ่านตั้งแต่ต้น) ⇒
แถวค้างอยู่ในรายการ แล้วป้าย 🔴 ใน §5.5 จะชี้ให้เห็นว่าชนกับชื่อไหนใน Odoo — **กับดัก 7.4 กลายเป็น
สิ่งที่ระบบมองเห็นแทนที่จะเป็นสิ่งที่ไม่มีใครรู้**

### 6.2 ธง `new_contact` บนใบ

```ts
// services/quotationService.ts — buildOdooManualReview()
if (enrichedQuote?.contact_pending_odoo) {
  const r = { kind: 'new_contact' as const, field: 'contact',   // คอลัมน์ C ของไฟล์ Odoo
              value: String(enrichedQuote?.customer_details?.contact_name ?? '') };
  reasons.push({ ...r, display_message: buildManualReasonDisplay(r) });
}
```

`buildOdooManualReview()` ถูกเรียก**ใน transaction ของ `confirmQuotationAtomic()`** และหัวฟังก์ชัน
เขียนกฎของตัวเองไว้ว่า *"อ่านจากข้อมูลของใบเองล้วน ๆ ไม่ยิง query เพิ่ม"* ⇒ `enrichQuotationData()`
(ซึ่งอยู่นอก tx เสมอ) เป็นคนเติมธง:

```
contact_pending_odoo = isLocalContactId(contact_id)
                       && (SELECT odoo_matched_at FROM local_contacts WHERE contact_id = …) IS NULL
```

* ยิง**เฉพาะเมื่อ `contact_id >= 900,000,000`** ⇒ ใบปกติทั้งหมดจ่าย 0 ms · เป็น PK lookup
* **fail-closed:** lookup ล้ม → ถือว่ายังไม่เข้า Odoo → ติดธง (ธงเกิน = เสียเวลา · ธงขาด = ของตกที่ Odoo)
* **ผูกกับ "ใบ" ไม่ใช่ "endpoint"** ตามกฎถาวรของระบบ ⇒ ใบจาก LINE ก็ติดธงเหมือนกัน (§4.1)

| มีสถานะให้ใครต้องมาปิด | — | ไม่มี · `odoo_exported_at` มาร์กเอง ใบออกจากคิวเอง |
| โค้ดที่ต้องเขียน | dialog + endpoint นับล่วงหน้า + แก้จุดกด export | ~10 บรรทัดใน `buildOdooManualReview()` |

### 6.4 ของที่ปุ่มติ๊กมือให้ไม่ได้ — **เมนูส่งออกรู้เองว่าพร้อมหรือยัง**

กลุ่ม `new_contact` ในเมนูส่งออกแสดง **2 ตัวเลข**: *พร้อมส่งออก* (ผู้ติดต่อของใบเข้า Odoo แล้ว) และ
*ยังคีย์ไม่เสร็จ* — คำนวณสดด้วย join `local_contacts` ด้วย `q.contact_id`
**ค่าตั้งต้นของปุ่มคือส่งออกเฉพาะที่พร้อม** (เลือกส่งทั้งหมดได้)
⇒ แอดมินไม่มีทางส่งใบออกไปก่อนที่ผู้ติดต่อจะมีอยู่จริงในปลายทางโดยไม่รู้ตัว

## 7. กับดัก 7 ข้อ

**7.1 `customers` ห้ามแตะ** — sync ทับทุกรอบ ของที่เขียนลงไปหายเงียบ นี่คือเหตุผลทั้งหมดที่โมดูลนี้
มีตารางของตัวเอง

**7.2 self-healing ทำงานเอง ห้ามเขียนโค้ดช่วย** — Odoo สร้างคนนี้จริงเมื่อไหร่ sync พาแถว
`source='odoo'` (คนละ `contact_id`) กลับมา แล้ว `local_taken` ทำให้แถว local หลบให้เอง ·
**เก็บแถวใน `local_contacts` ไว้ ไม่ต้องลบ** — มันคือหลักฐานว่าใบเก่าที่อ้าง `contact_id` 900 ล้าน
เคยหมายถึงใคร และเป็นที่เก็บ `odoo_matched_contact_id` ที่โยงสองโลกเข้าด้วยกัน

**7.3 ชื่อผู้ติดต่อใหม่ trim ตั้งแต่บันทึก — ไม่ขัดกับกฎห้าม trim** กฎ *"ห้าม `.trim()` ชื่อลูกค้า/
ผู้ติดต่อในทุกเส้นที่ส่งไป Odoo"* คุ้มครองชื่อที่ **Odoo เป็นเจ้าของ** (มีช่องว่างท้ายจริง 17,666 แถว)
ส่วนคนที่เราสร้างเอง เราเป็นต้นทาง — เก็บช่องว่างที่มองไม่เห็นไว้ให้คนคัดลอกไปวาง = สร้างเคส
"ชื่อไม่ตรง" ขึ้นมาเปล่า ๆ · เส้นทาง export ยังใช้ `cleanName()` ไม่ใช่ `clean()` เหมือนเดิมทุกประการ

**7.4 คีย์ชื่อใน Odoo ไม่ตรงเป๊ะ = ผู้ติดต่อซ้ำ + ใบนำเข้าไม่ได้** เว้นวรรคเกิน · เติม "คุณ" นำหน้า ·
สะกดต่าง → ทั้ง `local_taken` และสัญญาณ A/B ไม่ตรง → แถว local ไม่ยอมหลบ → **โผล่ซ้ำ 2 แถวใน
picker** และ Odoo จับคู่ผู้ติดต่อของใบไม่ได้
**ตัวรับมือในเฟสนี้:** ป้าย 🔴 "น่าจะคีย์แล้วแต่ชื่อไม่ตรง" (§5.5) · ปุ่ม "ใช้ชื่อนี้" สำหรับแถวที่ยังไม่มี
ใบอ้างถึง · และ **เคส diag ที่ยืนยันว่าชื่อไม่ตรงได้ 2 แถว** เพื่อให้พฤติกรรมนี้เป็นสิ่งที่ *รู้* ไม่ใช่
*เซอร์ไพรส์*

**7.5 ลบผู้ติดต่อที่มีใบอ้างอยู่ = `409`** ตรวจ `quotations.contact_id` ก่อนเสมอ · ลบได้ต้องลบทั้ง
`local_contacts` และแถวใน `customers_data_view` ใน transaction เดียว

**7.6 เฟส H (ตรึงทีมขาย) ต้องเสร็จก่อน — เจ้าของเคาะแล้ว · เขียนเสร็จบนเครื่อง dev 2026-09-18 ·
merge เข้า branch `dev` 2026-09-21**
คอลัมน์ I ของไฟล์ Odoo **เคย**มาจาก join **สด**
กับ `customers_data_view` ด้วย `contact_id` ([repositories.ts:471](../db/repositories.ts#L471))
พอ Odoo สร้างผู้ติดต่อจริงแล้ว sync กลับมา แถว local หายจาก view ⇒ **คอลัมน์ I ว่าง** และเพราะ
ลำดับงานคือ "ออกใบก่อน แล้วค่อยคีย์เข้า Odoo" เสมอ นี่จึงเป็น **ผลลัพธ์ปกติของทุกใบที่ออกให้
ผู้ติดต่อใหม่ ไม่ใช่เคสหายาก** · ทีมขายที่ถูกตรึงจะเป็นค่าที่สืบทอดจากบริษัทตาม §3.2
⇒ ตั้งแต่เฟส H ค่าถูกตรึงลง `quotations.customer_sales_team` ตอนยืนยัน และ export อ่าน

`scripts/dev/seedPhaseH.ts` ทำไว้แล้ว หยิบไปใช้ต่อได้เลย

**7.7 ผลการจับคู่ลูกค้าต้องไม่ขยับ** — `findCustomerCandidates` ค้นจาก `customers_data_view`
⇒ ผู้ติดต่อใหม่เปลี่ยนผลค้นได้ · cache ค้นหาไม่ต้องล้าง (เป็น `DISTINCT ON (company_id)` = รายชื่อ

**ด่านใหม่ `npm run diag:local-contacts` ต้องพิสูจน์อย่างน้อย 9 ข้อ**

1. **ตารางว่าง ⇒ `customers_data_build` ให้ผลเท่าเดิมทุกบิต** — เทียบจำนวนแถว + checksum ทุกคอลัมน์
2. **เวลา build ไม่ถอย** — วัด `CREATE TABLE AS SELECT * FROM customers_data_build` ก่อน–หลัง
   (ฐานอ้างอิง ~2–3 วิ) · ข้อนี้คือตัวจับ `LATERAL` ที่ §3.2 เตือนไว้
3. **แถวที่ dual-write เขียน = แถวที่ Arm 3 สร้างตอน rebuild — เทียบทีละคอลัมน์ทั้ง 24 ช่อง**
   (เพิ่ม → snapshot → rebuild `{force:true}` → เทียบ) ตัวจับอาการ "ค่า propagate ไม่ครบ" ·
   "`last_order_at` ไม่ตรงกับพี่น้อง" · **"ทีมขายสองฝั่งคนละกติกา"** ซึ่งทั้งสามไม่มีอะไรฟ้องจนกว่าจะออกใบผิด
4. **ทีมขายสืบทอดถูกทั้งสองทาง** — (ก) บริษัทที่แถวตัวแทนมีทีมว่างแต่พี่น้องมีค่า (663 บริษัท)
   ผู้ติดต่อใหม่ต้องได้ค่าของพี่น้อง **ไม่ใช่ค่าว่างของแถวตัวแทน** · (ข) บริษัทที่ไม่มีทีมขายเลย
   ต้องได้ `NULL` และ **`INSERT` ต้องผ่านไม่ใช่ล้ม** (§3.5 ข้อ 3 — ตัวจับคนที่เผลอใส่ `NOT NULL`
   หรือหาค่า default มาเติมให้ "ดูดีขึ้น")
5. ผู้ติดต่อใหม่ **โผล่ใน `getContactsByCustomerId` / `getContactById` / reverse lookup ทันที**
6. **ชื่อซ้ำ → `409`** และแถวเดิมถูกคืนมาให้เลือก · **บริษัทไม่มีจริง → `400`**
7. **`odoo_matched_at` ถูกเขียน ⟺ แถว local ถูกซ่อนจาก view** (สัญญาณ A ≡ `local_taken`)
8. **สัญญาณ B ยิงจากใบที่นำเข้าแล้ว** และ **ไม่ยิง**เมื่อ `sale_orders.contact_name` ไม่ตรงกับชื่อเรา
9. **ชื่อที่คีย์ใน Odoo ไม่ตรงเป๊ะ → ได้ 2 แถว + ป้าย 🔴 ชี้ชื่อที่ชน** (§7.4 — ยืนยันว่ารู้ตัว)

## 10. ทางที่ไม่ได้เลือก

| ทาง | ทำไมไม่เอา |
| --- | --- |
| **ปุ่ม ✅ ให้คนติ๊กว่าคีย์เข้า Odoo แล้ว** | คนคีย์ชื่อผิดแล้วติ๊กว่าเสร็จ = ระบบเชื่อ → ใบหลุดเข้าไฟล์ปกติ → ตกที่ Odoo โดยไม่มีใครรู้สาเหตุ · **การติ๊กมือกลบกับดัก 7.4 ทั้งหมด** ส่วนสัญญาณอัตโนมัติวัดจากของที่ Odoo ส่งกลับมาจริง จึงโกหกแบบนั้นไม่ได้ |
| ปุ่มคัดลอกรายช่องในตาราง | เจ้าของเลือกให้ทำงานจากไฟล์ xlsx/csv ทีเดียวทั้งชุด — ปุ่มคัดลอกทีละช่องเป็นงานทีละคนซึ่งช้ากว่า |
| เขียนลง `customers` ตรง ๆ | sync ทับทุกรอบ หายเงียบ |
| `REFRESH` ทั้งตารางทุกครั้งที่เพิ่มคน | build+swap ~2–3 วิ + `AccessExclusiveLock` · เพิ่มคน 1 คนไม่ควรจ่ายราคานั้น และไม่ควรแย่งจังหวะกับรอบ sync |
| ใช้ `contact_id` = 0 หรือติดลบ | 0 ถูกจองให้แถวบริษัทที่ไม่มีผู้ติดต่อ (3,852 แถว) · ติดลบถูกตัวกรอง `contact_id > 0` ทิ้งทั้งหมด (§1) |
| ย้าย `sales_team` เข้า CTE `comp` เพื่อ "ให้เหมือนกัน" | `comp` ตั้งใจไม่เอาทีมขายเข้า เพราะผู้ติดต่อคนละคนอาจคนละทีม — การสืบทอดใช้ได้เฉพาะกับคนใหม่ที่ไม่มีประวัติ |
| **ช่องให้แอดมินเขียนทับทีมขาย** (เคยร่างไว้ แล้วเจ้าของตัดออก 2026-09-17) | ปลายทางของช่องนี้คือคอลัมน์ I ของไฟล์ *ใบเสนอราคา* ซึ่งเฟส H ตรึงไปแล้ว ส่วนไฟล์ที่โมดูลนี้ผลิตคือ **template นำเข้าผู้ติดต่อคนละตัว ที่ไม่มีช่องทีมขาย** ⇒ เป็นช่องที่แอดมินต้องกรอกโดยไม่มีอะไรปลายทาง · และมันจะพ่วงของตามมาอีกชุด (คอลัมน์เก็บค่า · endpoint รายการทีม · การตรวจค่า · กติกา "ตรงกับค่าสืบทอด ⇒ เก็บ NULL") เพื่อแก้ปัญหาที่ไม่ได้เกิดจากโมดูลนี้ |
| เติมค่า default ให้ทีมขายที่ว่าง (59.1% ของบริษัท) | ค่าว่างคือ **ข้อเท็จจริงของฐาน Odoo** ไม่ใช่ข้อมูลขาดของเรา — เดาค่าให้แล้วใบจะวิ่งเข้าทีมที่ไม่ได้ดูแลลูกค้ารายนั้นจริง ซึ่งแย่กว่าว่าง |
| ตัวกรอง `source <> 'local'` ซ่อนผู้ติดต่อใหม่จากฝั่ง LINE | เจ้าของเคาะแล้วว่าให้เซลส์ออกใบให้คนนั้นได้ — และการซ่อนจะทำให้ "ผู้ติดต่อโผล่ครบทุกที่" ซึ่งเป็นหัวใจของ Arm 3 พังลงครึ่งหนึ่ง |
| `pg_advisory_lock` กันแข่งกับ build+swap | ปิดช่องว่างได้สนิทกว่า แต่ไปแตะเส้นทาง sync ซึ่งห้าม throw · `ensureDirectoryRow()` สองจุด (§3.4) ถูกกว่าและอยู่ในโมดูลของตัวเอง — ถ้าวันไหนวัดแล้วยังหลุด ค่อยกลับมาทางนี้ |
| pre-flight dialog ตอน export (แผนใหญ่ §5.6) | คิวแก้มือทำแทนได้ครบและไม่ต้องให้ใครตัดสินใจตอนกด (§6.3) |
| ตาราง log การคีย์เข้า Odoo แยก | audit trigger บน `local_contacts` บันทึกให้อยู่แล้ว รวมถึงตอนที่ **ระบบ** เป็นคนเขียน |
